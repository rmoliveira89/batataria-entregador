IMPORTANTE: Criar tabelas no Supabase
Para bloqueio e histórico funcionarem, crie no SQL Editor:
-- Tabela de reservas create table if not exists reservas_entregadores ( pedido_id text primary key, entregador_nome text, data text, created_at timestamp default now() ); alter table reservas_entregadores enable row level security; create policy "libera tudo anon" on reservas_entregadores for all using (true) with check (true); -- Tabela de confirmações (código pode ser NULL no modo entregador) create table if not exists confirmacoes_entregas ( id uuid primary key default gen_random_uuid(), numero_pedido text not null, codigo_cliente text, localizador text, nome_entregador text not null, data text not null, hora text not null, modo text not null, cliente_nome text, loja text, created_at timestamp default now() ); alter table confirmacoes_entregas enable row level security; create policy "libera tudo anon 2" on confirmacoes_entregas for all using (true) with check (true); -- Tabela de configuração global (modo para todos) create table if not exists app_config ( chave text primary key, valor text not null, atualizado_em timestamp default now() ); alter table app_config enable row level security; create policy "libera tudo anon 3" on app_config for all using (true) with check (true); -- Valor inicial insert into app_config (chave, valor) values ('modo_confirmacao','automatico') on conflict (chave) do nothing;

2. Botão 📋 COPIAR copia para clipboard
3. Botão "Abrir iFood" copia e abre portal do iFood
4. Lá no iFood ele cola o localizador e confirma a entrega
